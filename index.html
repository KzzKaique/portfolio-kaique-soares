PostgreSQL relatorio_sla_entregas.sql 
-- Análise de performance de entregas por região e período
-- PostgreSQL - Relatório de SLA de Entrega
WITH entregas_periodo AS (
    SELECT 
        r.nome_regiao,
        DATE_TRUNC('month', e.data_entrega) AS mes,
        COUNT(*) AS total_entregas,
        COUNT(CASE WHEN e.data_entrega <= e.data_prevista THEN 1 END) AS no_prazo,
        AVG(EXTRACT(EPOCH FROM (e.data_entrega - e.data_envio)) / 3600) AS tempo_medio_horas,
        SUM(e.valor_frete) AS receita_frete
    FROM entregas e
    INNER JOIN regioes r ON e.regiao_id = r.id
    WHERE e.data_entrega BETWEEN '2024-01-01' AND '2024-12-31'
        AND e.status = 'ENTREGUE'
    GROUP BY r.nome_regiao, DATE_TRUNC('month', e.data_entrega)
),
ranking AS (
    SELECT *,
        ROUND(100.0 * no_prazo / NULLIF(total_entregas, 0), 2) AS taxa_sla,
        RANK() OVER (PARTITION BY mes ORDER BY total_entregas DESC) AS rank_volume
    FROM entregas_periodo
)
SELECT 
    nome_regiao,
    TO_CHAR(mes, 'YYYY-MM') AS periodo,
    total_entregas,
    taxa_sla || '%' AS sla_entregas,
    ROUND(tempo_medio_horas, 1) || 'h' AS tempo_medio,
    TO_CHAR(receita_frete, 'R$ FM999G999D00') AS receita
FROM ranking
WHERE rank_volume <= 5
ORDER BY mes DESC, total_entregas DESC;
Oracle acuracia_inventario.sql 
-- Oracle - Análise de Acurácia de Inventário por Armazém
SELECT 
    a.codigo_armazem,
    a.nome_armazem,
    COUNT(DISTINCT i.sku) AS total_skus,
    SUM(CASE WHEN i.qtd_fisica = i.qtd_sistema THEN 1 ELSE 0 END) AS skus_acurados,
    ROUND(100 * SUM(CASE WHEN i.qtd_fisica = i.qtd_sistema THEN 1 ELSE 0 END) 
        / COUNT(DISTINCT i.sku), 2) AS acuracia_percentual,
    SUM(ABS(i.qtd_fisica - i.qtd_sistema) * p.custo_unitario) AS divergencia_valor,
    RANK() OVER (ORDER BY ROUND(100 * SUM(CASE WHEN i.qtd_fisica = i.qtd_sistema THEN 1 ELSE 0 END) 
        / COUNT(DISTINCT i.sku), 2) DESC) AS ranking_acuracia
FROM inventario i
INNER JOIN armazens a ON i.armazem_id = a.id
INNER JOIN produtos p ON i.sku = p.sku
WHERE i.data_contagem >= ADD_MONTHS(SYSDATE, -3)
GROUP BY a.codigo_armazem, a.nome_armazem
ORDER BY acuracia_percentual DESC;
Python etl_logistica.py 
"""
Script de automação para ETL de dados logísticos
Extrai dados de múltiplas fontes, transforma e carrega em PostgreSQL
"""
import pandas as pd
import psycopg2
from sqlalchemy import create_engine
from datetime import datetime, timedelta
import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')
logger = logging.getLogger(__name__)

class ETLLogistica:
    """Pipeline ETL para dados de operações logísticas"""
    
    def __init__(self, config: dict):
        self.engine = create_engine(
            f"postgresql://{config['user']}:{config['password']}"
            f"@{config['host']}:{config['port']}/{config['database']}"
        )
        self.data_referencia = datetime.now()
        logger.info("Pipeline ETL inicializado com sucesso")
    
    def extrair_dados_entregas(self, dias_retroativos: int = 30) -> pd.DataFrame:
        """Extrai dados de entregas dos últimos N dias"""
        query = """
            SELECT e.*, c.nome_cliente, c.segmento,
                   r.nome_regiao, r.uf
            FROM entregas e
            JOIN clientes c ON e.cliente_id = c.id
            JOIN regioes r ON e.regiao_id = r.id
            WHERE e.data_entrega >= %(data_inicio)s
        """
        params = {
            'data_inicio': self.data_referencia - timedelta(days=dias_retroativos)
        }
        df = pd.read_sql(query, self.engine, params=params)
        logger.info(f"Extraídos {len(df)} registros de entregas")
        return df
    
    def transformar_indicadores(self, df: pd.DataFrame) -> pd.DataFrame:
        """Calcula KPIs de performance logística"""
        df['entrega_no_prazo'] = df['data_entrega'] <= df['data_prevista']
        df['tempo_entrega_horas'] = (
            (df['data_entrega'] - df['data_envio']).dt.total_seconds() / 3600
        )
        
        indicadores = df.groupby(['nome_regiao', 'segmento']).agg(
            total_entregas=('id', 'count'),
            entregas_no_prazo=('entrega_no_prazo', 'sum'),
            tempo_medio_horas=('tempo_entrega_horas', 'mean'),
            receita_total=('valor_frete', 'sum')
        ).reset_index()
        
        indicadores['taxa_sla'] = round(
            100 * indicadores['entregas_no_prazo'] / indicadores['total_entregas'], 2
        )
        
        logger.info(f"Gerados indicadores para {len(indicadores)} combinações região/segmento")
        return indicadores
    
    def carregar_dados(self, df: pd.DataFrame, tabela: str):
        """Carrega dados transformados no banco de dados"""
        df.to_sql(tabela, self.engine, if_exists='replace', index=False)
        logger.info(f"Carregados {len(df)} registros na tabela '{tabela}'")

if __name__ == "__main__":
    config = {

// Taxa de SLA de Entrega - Medida com contexto de filtro
Taxa SLA Entrega = 
VAR _TotalEntregas = COUNTROWS(Entregas)
VAR _NoPrazo = 
    CALCULATE(
        COUNTROWS(Entregas),
        Entregas[DataEntrega] <= Entregas[DataPrevista]
    )
RETURN
    DIVIDE(_NoPrazo, _TotalEntregas, 0)

// Acurácia de Inventário Ponderada por Valor
Acurácia Inventário = 
VAR _ValorCorreto = 
    SUMX(
        FILTER(Inventario, Inventario[QtdFisica] = Inventario[QtdSistema]),
        Inventario[QtdFisica] * RELATED(Produtos[CustoUnitario])
    )
VAR _ValorTotal = 
    SUMX(
        Inventario,
        Inventario[QtdSistema] * RELATED(Produtos[CustoUnitario])
    )
RETURN
    DIVIDE(_ValorCorreto, _ValorTotal, 0)

    // Calcular divergência
    DivergenciaAdicionada = Table.AddColumn(
        ColunasRemovidas, 
        "Divergencia", 
        each [QtdFisica] - [QtdSistema], 
        type number
    ),
    
    // Classificar tipo de divergência
    TipoDivergencia = Table.AddColumn(
        DivergenciaAdicionada,
        "TipoDivergencia",
        each if [Divergencia] > 0 then "Sobra"
            else if [Divergencia] < 0 then "Falta"
            else "Acurado",
        type text
    ),
    
    // Classificação ABC por valor
    ClassificacaoABC = Table.AddColumn(
        TipoDivergencia,
        "CurvaABC",
        each if [PercentualAcumulado] <= 0.8 then "A"
            else if [PercentualAcumulado] <= 0.95 then "B"
            else "C",
        type text
    ),
    
    TiposDefinidos = Table.TransformColumnTypes(
        ClassificacaoABC,
